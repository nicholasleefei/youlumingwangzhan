import React, { useState, useEffect } from 'react';
import { supabase } from '@/utils/supabaseClient';

export default function BrandFilter({ onBrandChange }: { onBrandChange: (brand: string | null) => void }) {
  const [brands, setBrands] = useState<string[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchBrands() {
      const { data, error } = await supabase
        .from('models')
        .select('brand')
        .eq('is_active', true)
        .distinct('brand');
      
      if (error) {
        console.error('Error fetching brands:', error);
        return;
      }
      
      const brandList = (data || []).map(item => item.brand).filter(Boolean) as string[];
      setBrands(brandList);
      setLoading(false);
    }
    
    fetchBrands();
  }, []);

  if (loading) return <div>Loading brands...</div>;

  return (
    <select 
      onChange={(e) => onBrandChange(e.target.value === 'all' ? null : e.target.value)}
      className="rounded-lg border border-zinc-200 bg-white px-3 py-2 text-sm text-zinc-900"
    >
      <option value="all">All Brands</option>
      {brands.map(brand => (
        <option key={brand} value={brand}>{brand}</option>
      ))}
    </select>
  );
}

import React, { useState } from 'react';

export default function PriceFilter({ onPriceChange }: { onPriceChange: (min: number | null, max: number | null) => void }) {
  const [minPrice, setMinPrice] = useState('');
  const [maxPrice, setMaxPrice] = useState('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    onPriceChange(
      minPrice ? parseInt(minPrice) : null,
      maxPrice ? parseInt(maxPrice) : null
    );
  };

  return (
    <form onSubmit={handleSubmit} className="flex items-center gap-2">
      <input
        type="number"
        value={minPrice}
        onChange={(e) => setMinPrice(e.target.value)}
        placeholder="Min Price"
        className="w-32 rounded-lg border border-zinc-200 bg-white px-3 py-2 text-sm text-zinc-900"
      />
      <span className="text-zinc-500">-</span>
      <input
        type="number"
        value={maxPrice}
        onChange={(e) => setMaxPrice(e.target.value)}
        placeholder="Max Price"
        className="w-32 rounded-lg border border-zinc-200 bg-white px-3 py-2 text-sm text-zinc-900"
      />

        type="submit"
        className="rounded-lg bg-amber-500 px-3 py-2 text-sm font-medium text-zinc-950 hover:bg-amber-400"

        Filter
      </button>
    </form>
  );
}

import React, { useState } from 'react';

export default function FileUpload({ onFilesChange }: { onFilesChange: (files: File[]) => void }) {
  const [files, setFiles] = useState<File[]>([]);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files) {
      const newFiles = Array.from(e.target.files);
      setFiles([...files, ...newFiles]);
      onFilesChange([...files, ...newFiles]);
    }
  };

  return (
    <div className="space-y-2">
      <input
        type="file"
        multiple
        onChange={handleFileChange}
        className="w-full rounded-lg border border-zinc-200 bg-white px-3 py-2 text-sm text-zinc-900"
      />
      {files.length > 0 && (
        <div className="space-y-1">
          {files.map((file, index) => (
            <div key={index} className="flex items-center justify-between rounded-lg bg-zinc-50 px-3 py-2 text-sm text-zinc-700">
              <span>{file.name}</span>
              <span>{(file.size / 1024).toFixed(2)} KB</span>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

// src/pages/SalesStats.tsx
import React, { useState, useEffect } from 'react';
import { supabase } from '@/utils/supabaseClient';

export default function SalesStats() {
  const [sales, setSales] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchSales() {
      const { data, error } = await supabase
        .from('sales_statistics')
        .select('*')
        .order('period', { ascending: false });
      
      if (error) {
        console.error('Error fetching sales:', error);
        return;
      }
      
      setSales(data || []);
      setLoading(false);
    }
    
    fetchSales();
  }, []);

  if (loading) return <div>Loading sales data...</div>;

  return (
    <div className="mx-auto max-w-6xl px-4 py-10">
      <h1 className="text-2xl font-semibold text-zinc-900">Sales Statistics</h1>
      <div className="mt-6 grid gap-6 md:grid-cols-2 lg:grid-cols-3">
        {sales.map(item => (
          <div key={item.id} className="rounded-xl border border-zinc-200 bg-white p-6">
            <h3 className="font-medium text-zinc-900">{item.country}</h3>
            <p className="text-sm text-zinc-600">{item.period}</p>
            <div className="mt-4 flex items-end justify-between">
              <div>
                <p className="text-sm text-zinc-500">Brand</p>
                <p className="font-semibold text-zinc-900">{item.brand}</p>
              </div>
              <div>
                <p className="text-sm text-zinc-500">Quantity</p>
                <p className="font-semibold text-zinc-900">{item.quantity}</p>
              </div>
              <div>
                <p className="text-sm text-zinc-500">Revenue</p>
                <p className="font-semibold text-zinc-900">${item.revenue.toLocaleString()}</p>
              </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
